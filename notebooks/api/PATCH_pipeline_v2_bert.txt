1c. Ajouter une méthode de chargement et de prédiction BERT dans
    HybridEmailPipelineV2, juste après _predict_ml :

    def load_bert(self):
        """Charge DistilBERT si le dossier du modèle existe. Optionnel :
        le pipeline continue de fonctionner sans BERT si absent (comme
        avant), simplement avec un poids bert nul redistribué vers ml."""
        if not BERT_MODEL_DIR.exists():
            logger.warning(f"DistilBERT introuvable dans {BERT_MODEL_DIR} — "
                            "pipeline exécuté sans couche BERT.")
            self._bert_tokenizer = None
            self._bert_model = None
            return
        logger.info("Chargement de DistilBERT...")
        self._bert_tokenizer = DistilBertTokenizerFast.from_pretrained(str(BERT_MODEL_DIR))
        self._bert_model = DistilBertForSequenceClassification.from_pretrained(
            str(BERT_MODEL_DIR), num_labels=len(self._class_names)
        ).to(BERT_DEVICE)
        self._bert_model.eval()
        logger.info(f"DistilBERT chargé sur {BERT_DEVICE}.")

    @torch.no_grad()
    def _predict_bert(self, text: str) -> dict:
        if self._bert_model is None:
            return {}
        encoding = self._bert_tokenizer(
            text[:2000], max_length=BERT_MAX_LENGTH,
            padding='max_length', truncation=True, return_tensors='pt',
        )
        outputs = self._bert_model(
            input_ids=encoding['input_ids'].to(BERT_DEVICE),
            attention_mask=encoding['attention_mask'].to(BERT_DEVICE),
        )
        probs = torch.softmax(outputs.logits, dim=-1).cpu().numpy()[0]
        # IMPORTANT : mapper par NOM de classe, pas par index — l'ordre
        # interne de DistilBertForSequenceClassification (id2label) doit
        # être recoupé avec self._class_names (celui du label_encoder du
        # RF), qui peuvent différer selon l'ordre d'entraînement.
        id2label = self._bert_model.config.id2label
        bert_proba_by_name = {id2label[i]: float(p) for i, p in enumerate(probs)}
        return {c: bert_proba_by_name.get(c, 0.0) for c in self._class_names}

────────────────────────────────────────────────────────────────────
CHANGEMENT 2 — Protéger le signal RF/BERT fort contre un correcteur
                mal calibré
────────────────────────────────────────────────────────────────────

Dans _apply_corrector(), le poids du correcteur (w_corr) peut aujourd'hui
diluer une prédiction RF déjà confiante sur la classe phishing — c'est la
cause directe du F1-Macro à 0,62 observé. On ajoute une garde : si le RF
est déjà très confiant sur 'phishing', on réduit fortement l'influence du
correcteur pour cette prédiction précise, quelle que soit sa propre
confiance.

Avant (fin de _apply_corrector) :

    # Poids de base
    w_corr = 0.4 * corr_conf

    # Boost spam : si le correcteur détecte du spam avec certitude,
    # augmenter son poids car RF sous-prédit structurellement cette classe
    if corr_pred == 'spam' and corr_conf > 0.50:
        w_corr = min(w_corr + 0.15, 0.55)
        logger.debug(f"Spam boost activé (w_corr={w_corr:.2f})")

    w_rf = 1.0 - w_corr
    final = w_rf * rf_proba + w_corr * corr_proba
    final = final / final.sum()

    return {c: float(p) for c, p in zip(class_names, final)}

Après (remplace ce bloc) :

    # Poids de base
    w_corr = 0.4 * corr_conf

    # Boost spam : si le correcteur détecte du spam avec certitude,
    # augmenter son poids car RF sous-prédit structurellement cette classe
    if corr_pred == 'spam' and corr_conf > 0.50:
        w_corr = min(w_corr + 0.15, 0.55)
        logger.debug(f"Spam boost activé (w_corr={w_corr:.2f})")

    # Garde de protection : si le RF est déjà très confiant sur phishing,
    # le correcteur ne doit pas pouvoir écraser ce signal — tant que le
    # correcteur n'a pas reçu un volume suffisant de vraies corrections
    # phishing équilibrées (cf. generate_balanced_corrections.py), son
    # jugement sur cette classe reste structurellement moins fiable que
    # celui du RF seul.
    rf_phish_idx = class_names.index('phishing') if 'phishing' in class_names else None
    if rf_phish_idx is not None and rf_proba[rf_phish_idx] > 0.70:
        w_corr = min(w_corr, 0.15)
        logger.debug(f"Protection RF phishing activée (w_corr plafonné à {w_corr:.2f})")

    w_rf = 1.0 - w_corr
    final = w_rf * rf_proba + w_corr * corr_proba
    final = final / final.sum()

    return {c: float(p) for c, p in zip(class_names, final)}
